# SNI allowlist (regex, 1 行 1 entry)
#
# 運用ルール:
#   - 新しい package registry を使うツールを足す PR では、この allowlist 差分も
#     同じ PR で更新する
#   - agent の WebFetch allowlist と documentation domain は同時更新する
#     (片方だけ更新しても効かない)
#   - write 系 (production への CREATE/UPDATE/DELETE) credential はこの allowlist
#     経由では渡さない (laptop 側に留める)

# ---- Anthropic (Claude Code API / OAuth / docs / bridge) ----
^.*\.anthropic\.com$
^claude\.ai$
^.*\.claude\.ai$
^platform\.claude\.com$
^.*\.claudeusercontent\.com$
^storage\.googleapis\.com$

# ---- OpenAI (codex API + ChatGPT OAuth) ----
# codex は自身の API 通信で api.openai.com、OAuth ログインで auth.openai.com /
# chatgpt.com 系に到達する。これらが無いと sandbox 内 codex は応答を得られない
# (OAuth ログイン自体は sandbox 外で実行)。
^api\.openai\.com$
^auth\.openai\.com$
^chatgpt\.com$
^.*\.chatgpt\.com$

# ---- GitHub (Claude Code / git / gh CLI) ----
# apex (github.com) は subdomain regex にマッチしないので別途許可する。
# mise 等が version 取得で git ls-remote https://github.com/... を叩く。
^github\.com$
^.*\.github\.com$
^.*\.githubusercontent\.com$
^.*\.githubassets\.com$
^codeload\.github\.com$
^raw\.githubusercontent\.com$

# ---- Package registries ----
^registry\.npmjs\.org$
^.*\.npmjs\.com$
^pypi\.org$
^.*\.pythonhosted\.org$
^mise\.jdx\.dev$
^mise-versions\.jdx\.dev$
# mise core:go backend が Go 本体を取得する配布元。
^dl\.google\.com$
# go mod / go build がモジュールを取得・検証する Go 公式インフラ。
# proxy.golang.org = module proxy (GOPROXY 既定。.info/.mod/.zip 配布)、
# sum.golang.org   = checksum database (GOSUMDB 既定。新規 module の go.sum 検証)。
# どちらも read-only な公開エンドポイント (publish/write 系は存在しない)。
^proxy\.golang\.org$
^sum\.golang\.org$
# mise core:rust backend が rustup 経由で Rust toolchain を取得する配布元。
# sh.rustup.rs = rustup-init bootstrap スクリプト、
# static.rust-lang.org = rustup-init 本体 + toolchain dist (/rustup, /dist) の CDN。
^sh\.rustup\.rs$
^static\.rust-lang\.org$
# cargo が依存 crate を取得する配布元 (sparse registry, cargo 1.70+ 既定)。
# index.crates.io = sparse index (crate メタデータ)、static.crates.io = crate tarball DL。
# apex crates.io (publish/search/yank の write・account API) は意図的に未許可。
^index\.crates\.io$
^static\.crates\.io$
# mise が aqua backend のツール (uv 等) を入れる際、GitHub artifact attestation を
# Sigstore で検証する。その TUF root metadata の配布元 (tuf-repo-cdn.sigstore.dev)。
^tuf-repo-cdn\.sigstore\.dev$

# ---- OS updates ----
^.*\.ubuntu\.com$
^.*\.canonical\.com$
^keyserver\.ubuntu\.com$

# ---- Tailscale (control plane / DERP relay / apt repo / client logs) ----
# tailscaled が coordination (controlplane./login.tailscale.com)、DERP relay
# (derpN.tailscale.com)、apt repo + 署名鍵 (pkgs.tailscale.com) に TCP/443 で到達する。
# いずれも *.tailscale.com に含まれる。WireGuard data plane は UDP/41641 のため redirect
# 対象外 (squid は通らない)。log.tailscale.io は client metrics/logs (既定 on)。
^.*\.tailscale\.com$
^log\.tailscale\.io$

# ---- AWS (SSM / CloudWatch + 別アカウント read-only 確認用、広め) ----
# 日常業務で別アカウントの read-only 確認をするため広めに許可。
# トレードオフ: sts/iam/s3.* 等も通る = exfil 経路は構造的に閉じきれない。
# write 系 credential を渡さない運用規律でカバーする。
^[a-z0-9-]+\.amazonaws\.com$
^[a-z0-9-]+\.[a-z0-9-]+\.amazonaws\.com$
# AWS CLI v2 既定の S3 regional virtual-hosted-style
# (<bucket>.s3.<region>.amazonaws.com = 3 ラベル) は上の 2 ラベル pattern で
# 拾えず terminate される (aws s3 ls が SSL WRONG_VERSION_NUMBER で失敗)。
# 別 region bucket の read 確認用に S3 vhost 形に限定して許可する。
^[a-z0-9-]+\.s3\.[a-z0-9-]+\.amazonaws\.com$

# ---- Documentation domains for agent WebFetch ----
# agent の WebFetch(domain:...) allowlist と必ず同時更新すること。
^docs\.aws\.amazon\.com$
^developer\.mozilla\.org$
^nodejs\.org$
^(.*\.)?react\.dev$
^pnpm\.io$
^docs\.github\.com$
