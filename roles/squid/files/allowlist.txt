# SNI allowlist (regex, 1 行 1 entry)
#
# 運用ルール:
#   - 新しい package registry を使うツールを足す PR では、この allowlist 差分も
#     同じ PR で更新する
#   - agent の WebFetch allowlist と documentation domain は同時更新する
#     (片方だけ更新しても効かない)
#   - write 系 (production への CREATE/UPDATE/DELETE) credential はこの allowlist
#     経由では渡さない (laptop 側に留める)

# ---- Anthropic (Claude Code API / OAuth / docs / bridge) ----
^.*\.anthropic\.com$
^claude\.ai$
^.*\.claude\.ai$
^platform\.claude\.com$
^.*\.claudeusercontent\.com$
^storage\.googleapis\.com$

# ---- OpenAI (codex API + ChatGPT OAuth) ----
# codex は自身の API 通信で api.openai.com、OAuth ログインで auth.openai.com /
# chatgpt.com 系に到達する。これらが無いと sandbox 内 codex は応答を得られない
# (OAuth ログイン自体は sandbox 外で実行)。
^api\.openai\.com$
^auth\.openai\.com$
^chatgpt\.com$
^.*\.chatgpt\.com$

# ---- GitHub (Claude Code / git / gh CLI) ----
^.*\.github\.com$
^.*\.githubusercontent\.com$
^.*\.githubassets\.com$
^codeload\.github\.com$
^raw\.githubusercontent\.com$

# ---- Package registries ----
^registry\.npmjs\.org$
^.*\.npmjs\.com$
^pypi\.org$
^.*\.pythonhosted\.org$
^mise\.jdx\.dev$

# ---- OS updates ----
^.*\.ubuntu\.com$
^.*\.canonical\.com$
^keyserver\.ubuntu\.com$

# ---- AWS (SSM / CloudWatch + 別アカウント read-only 確認用、広め) ----
# 日常業務で別アカウントの read-only 確認をするため広めに許可。
# トレードオフ: sts/iam/s3.* 等も通る = exfil 経路は構造的に閉じきれない。
# write 系 credential を渡さない運用規律でカバーする。
^[a-z0-9-]+\.amazonaws\.com$
^[a-z0-9-]+\.[a-z0-9-]+\.amazonaws\.com$

# ---- Documentation domains for agent WebFetch ----
# agent の WebFetch(domain:...) allowlist と必ず同時更新すること。
^docs\.aws\.amazon\.com$
^developer\.mozilla\.org$
^nodejs\.org$
^(.*\.)?react\.dev$
^pnpm\.io$
^docs\.github\.com$
